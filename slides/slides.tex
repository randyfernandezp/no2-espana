\begin{frame}{El problema está mal condicionado}

\begin{columns}[T]
\begin{column}{0.56\textwidth}
\ce{NO2} tiene \textbf{dos firmas simultáneas}:
\begin{itemize}
  \item emisiones puntuales industriales
  \item emisiones lineales de tráfico
\end{itemize}

\vspace{3mm}
Los modelos de referencia (ESCAPE, LUR europeos) alcanzan
$R^2 \approx \numrange{0.5}{0.8}$\dots

\vspace{2mm}
\textbf{\dots pero no separan mecanismos.}

\vspace{3mm}
{\color{gris}En la llanura padana o la cuenca del Ruhr, industria y ciudad
son la misma covariable.}
\end{column}

\begin{column}{0.42\textwidth}
\begin{block}{No es un problema de método}
Es de \textbf{identificabilidad}.

\vspace{2mm}
Ningún estimador puede separar dos covariables casi colineales.
\end{block}
\end{column}
\end{columns}

\end{frame}

\begin{frame}{Por qué España}

\begin{center}
\Large
Los grandes emisores están \textbf{lejos} de los grandes focos urbanos.
\end{center}

\vspace{4mm}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Focos industriales}\\[1mm]
Huelva \textperiodcentered\ Tarragona \textperiodcentered\ Puertollano\\
As Pontes \textperiodcentered\ Avilés \textperiodcentered\ Algeciras
\end{column}
\begin{column}{0.48\textwidth}
\textbf{Focos urbanos}\\[1mm]
Madrid \textperiodcentered\ Barcelona
\end{column}
\end{columns}

\vspace{6mm}
\begin{block}{Diseño natural}
Un país donde el efecto industrial \textbf{puede} estimarse sin confundirse
con la urbanización.
\end{block}

\vspace{2mm}
{\small\color{gris}Coste declarado: Canarias concentra el \SI{24}{\percent}
del \ce{NOx} industrial y queda fuera del dominio (se exige conexidad).}

\end{frame}

% =============================================================================
\begin{frame}{La pregunta y el diseño}

\begin{center}
\large
¿A qué \textbf{escala espacial} opera la influencia industrial?
\end{center}

\vspace{4mm}
Dos estimaciones \textbf{metodológicamente independientes}:

\vspace{3mm}
\begin{columns}[T]
\begin{column}{0.47\textwidth}
\begin{block}{Rama B --- proceso puntual}
Observa \textbf{solo coordenadas} de \num{297} instalaciones.

\vspace{1mm}
\emph{¿A qué distancia se agrupan entre sí?}
\end{block}
\end{column}
\begin{column}{0.47\textwidth}
\begin{block}{Rama C --- acoplamiento}
Observa \textbf{concentraciones} en \num{212} estaciones.

\vspace{1mm}
\emph{¿A qué distancia deja de notarse la emisión?}
\end{block}
\end{column}
\end{columns}

\vspace{5mm}
\begin{center}
\color{rojo}\textbf{No hay ninguna razón aritmética para que coincidan.}
\end{center}

\end{frame}

% =============================================================================
\begin{frame}{Tres decisiones que condicionan todo}

\begin{enumerate}
  \item[\textbf{1}] \textbf{Solo estaciones de fondo alimentan el modelo}
  ($n = \num{212}$ de \num{462}).

  {\small\color{gris}Las de tráfico miden a microescala: el \ce{NO2} cae
  \SIrange{60}{80}{\percent} en los primeros \SI{50}{\metre}. No son
  realizaciones del campo regional.}

  \vspace{1mm}
  {\small$\rightarrow$ Se reservan como \textbf{validación externa}. El sesgo
  medido en ellas \emph{cuantifica el recargo local}.}

  \vspace{4mm}
  \item[\textbf{2}] \textbf{Desagrupamiento} por celdas de
  \SI{25}{\kilo\metre}: media \SI{11.36}{\ugm} $\rightarrow$
  \SI{8.58}{\ugm}.

  \vspace{4mm}
  \item[\textbf{3}] \textbf{Respuesta areal, no exposición ponderada.}

  {\small\color{gris}La exposición ponderada correlaciona \num{0.948} con la
  densidad \emph{por construcción}. Regresarla sobre densidad sería circular.}
\end{enumerate}

\end{frame}

\begin{frame}{Rama B --- Dónde están las fuentes}

\num{297} instalaciones. Aleatoriedad espacial rechazada
($p = \num{0.002}$).

\vspace{3mm}
\centering
\begin{tabular}{@{}lcc@{}}
\toprule
\textbf{Efecto por rango intercuartílico} & \textbf{Factor} & $p$ \\
\midrule
Suelo industrial & $\times\num{3.94}$ & $<\num{0.001}$ \\
Superficie artificial & $\times\num{1.32}$ & $<\num{0.001}$ \\
\rowcolor{yellow!25}

\begin{block}{Lectura}
Condicionando en uso del suelo, \textbf{dónde vive la gente no predice dónde

\vspace{1mm}
Evidencia directa contra la hipótesis de colinealidad total.
\end{block}

\end{frame}

\begin{frame}{Rama C --- La escala de influencia}

Superficie de presión ponderada por emisión:
\[
P_h(\mathbf{u}) = \sum_i \log(1 + m_i)\,\kappa_h(\|\mathbf{u}-\mathbf{s}_i\|)
\]

Barrido de siete anchos de banda. La escala es \textbf{estimada}, no fijada
por diseño.

\vspace{3mm}
\begin{center}
\begin{tabular}{@{}lcccccc@{}}
\toprule
$h$ (km) & \num{5} & \textbf{10} & \num{15} & \num{25} & \num{40} & \num{100} \\
\midrule
$\Delta$AIC & \num{2.8} & \textbf{\num{0.0}} & \num{1.5} & \num{5.8} & \num{7.1} & \num{12.9} \\
\bottomrule
\end{tabular}
\end{center}

\vspace{3mm}
$F = \num{13.86}$, $p = \num{2.5e-4}$. \quad $R^2$ aj.\ \num{0.754}
$\rightarrow$ \num{0.769}.

\vspace{3mm}
{\small\color{gris}Lo que \emph{no} dice: la mejora del RMSE es del
\SI{0.41}{\percent}. Robusto pero de escaso poder predictivo.}

\end{frame}

\begin{frame}{El artefacto de \textsc{nuts}-2}

A \num{16} regiones aparece un coeficiente industrial
\textbf{negativo y significativo} ($\hat\beta = \num{-0.562}$,
$p = \num{0.006}$).

\vspace{3mm}
Diagnóstico:
\begin{itemize}
  \item Madrid: distancia de Cook $= \num{4.64}$
  {\small\color{gris}(19$\times$ el umbral, 17$\times$ la siguiente)}
  \item sin Madrid: $\hat\beta = \num{-0.063}$, $p = \num{0.68}$
  \item $R^2$ aj.\ \num{0.775} $\rightarrow$ \num{0.597}
  {\small\color{gris}(el de \textsc{nuts}-3 es \num{0.601})}
\end{itemize}

\vspace{3mm}
Sobre \num{1000} particiones alternativas: $\hat\beta \in
[\num{-0.146}, \num{0.003}]$.

\vspace{2mm}
\begin{alertblock}{}
\centering
El valor de \textsc{nuts}-2 queda \textbf{fuera del rango completo}
por un factor de casi cuatro.
\end{alertblock}

\vspace{2mm}
{\small\color{gris}Madrid \textbf{nunca} queda aislada en las mil réplicas:
\textsc{nuts}-2 la aísla por razones históricas, no geométricas.}

\end{frame}

\begin{frame}{Conclusiones}

\begin{enumerate}
  \item[\textbf{1}] Dos ramas independientes sitúan la escala industrial en
  \textbf{$\approx$\SI{10}{\kilo\metre}} (razón \num{0.93}).
  La unidad de influencia es el \textbf{distrito}, no la instalación.

  \vspace{3mm}
  \item[\textbf{2}] El mismo efecto es \textbf{indetectable} a escala
  provincial. No por falta de potencia: por la unidad de análisis.

  \vspace{3mm}
  \item[\textbf{3}] El MAUP pasa de advertencia genérica a
  \textbf{diagnóstico cuantitativo} cuando existe una escala de referencia
  estimada externamente.
\end{enumerate}

\vspace{5mm}
\begin{center}
\color{azul}\large
Sin la rama de procesos puntuales, el resultado nulo areal sería\\
indistinguible de una ausencia real de efecto.
\end{center}

\end{frame}

% =============================================================================
\begin{frame}{Limitaciones}

\begin{itemize}\small
  \item \textbf{Umbral E-PRTR \SI{100}{\tonne} anuales} --- el patrón
  representa grandes emisores, no toda fuente industrial.
  \item \textbf{Canarias fuera del dominio} con el \SI{24}{\percent} del
  \ce{NOx} industrial español. Conclusiones válidas para península y Baleares.
  \item \textbf{La superficie modela fondo, no exposición máxima} --- los
  picos de tráfico ocurren por debajo de la resolución de
  \SI{2}{\kilo\metre}.
  \item \textbf{CORINE clase 121} mezcla industria con comercio y logística;
  la señal industrial limpia es el patrón E-PRTR.
  \item \textbf{$n = \num{50}$ unidades areales} --- poca potencia; el
  diagnóstico de influyentes lo confirma.
  \item \textbf{Envolventes con $\lambda$ fijada} de los propios datos:
  contraste ligeramente anticonservador.
\end{itemize}

\vspace{3mm}
{\small\textbf{Trabajo futuro:} formulación bayesiana con SPDE para propagar
la incertidumbre de la superficie hacia el modelo areal; extensión temporal
2019--2023 usando cierres de instalaciones como cuasiexperimento.}

\end{frame}
